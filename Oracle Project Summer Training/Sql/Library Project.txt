CREATE TABLE REGISTRATION( NAME VARCHAR2(20),DATED DATE,R_ID INT PRIMARY KEY);

CREATE TABLE DEPARTMENT( D_ID NUMBER(10) PRIMARY KEY,D_NAME VARCHAR2(10));

CREATE TABLE STUDENT( S_ID NUMBER(20) PRIMARY KEY,S_NAME VARCHAR2(20),D_ID REFERENCES DEPARTMENT,CONTACT_NO INTEGER,
YEAR INTEGER,R_ID REFERENCES REGISTRATION);

CREATE TABLE STAFF(ST_ID VARCHAR2(10) PRIMARY KEY,ST_NAME VARCHAR2(10),DESIGNATION VARCHAR2(20),R_ID REFERENCES REGISTRATION);

CREATE TABLE BOOKSELF( B_ID NUMBER(5) PRIMARY KEY,B_TITLE VARCHAR2(20),B_AUTHOR VARCHAR2(20),D_ID REFERENCES DEPARTMENT,
NO_OF_COPIES NUMBER(20),NO_OF_ISSUED_BOOK NUMBER(20));

CREATE TABLE ISSUE( B_ID REFERENCES BOOKSELF,ST_ID REFERENCES STAFF, S_ID REFERENCES STUDENT,ISSUING DATE,SUBMITTING DATE);
